```python
"""
게임 영상 이벤트 밀집 구간 자동 추천 스크립트
사용법: python segment_selector.py <youtube_url_or_id> [--duration 60] [--window 5] [--hop 1] [--min_peaks 3] [--top 10]
"""

import argparse
import subprocess
import os
import sys
import numpy as np
import librosa
from sklearn.metrics.pairwise import cosine_similarity
from scipy.signal import find_peaks
import json


def download_audio(youtube_id, output_path="temp_audio.wav"):
    """YouTube 영상에서 오디오 추출"""
    url = f"https://www.youtube.com/watch?v={youtube_id}"
    print(f"[1/5] 오디오 다운로드 중: {url}")
    
    cmd = [
        "yt-dlp",
        "-x", "--audio-format", "wav",
        "--audio-quality", "0",
        "-o", output_path.replace(".wav", ".%(ext)s"),
        url
    ]
    subprocess.run(cmd, check=True, capture_output=True)
    
    # yt-dlp가 생성한 파일 찾기
    base = output_path.replace(".wav", "")
    for ext in [".wav", ".webm", ".m4a", ".mp3"]:
        if os.path.exists(base + ext) and ext != ".wav":
            subprocess.run([
                "ffmpeg", "-y", "-i", base + ext,
                "-vn", "-ar", "22050", output_path
            ], check=True, capture_output=True)
            os.remove(base + ext)
            break
    
    if not os.path.exists(output_path):
        raise FileNotFoundError("오디오 다운로드 실패")
    
    print(f"   -> {output_path} 저장 완료")
    return output_path


def compute_novelty(ssm, kernel_size=16):
    """SSM에서 checkerboard 커널로 novelty score 계산"""
    kernel = np.ones((kernel_size, kernel_size))
    half = kernel_size // 2
    for i in range(kernel_size):
        for j in range(kernel_size):
            if (i < half and j < half) or (i >= half and j >= half):
                kernel[i, j] = 1
            else:
                kernel[i, j] = -1
    
    n = ssm.shape[0]
    novelty = np.zeros(n)
    for i in range(half, n - half):
        patch = ssm[i-half:i+half, i-half:i+half]
        novelty[i] = np.sum(patch * kernel)
    
    novelty = np.maximum(novelty, 0)
    if novelty.max() > 0:
        novelty /= novelty.max()
    return novelty


def compute_ssm_chunked(segments, chunk_size=1000):
    """메모리 절약을 위해 SSM을 청크 단위로 계산"""
    n = len(segments)
    ssm = np.zeros((n, n), dtype=np.float32)
    for i in range(0, n, chunk_size):
        i_end = min(i + chunk_size, n)
        for j in range(0, n, chunk_size):
            j_end = min(j + chunk_size, n)
            ssm[i:i_end, j:j_end] = cosine_similarity(segments[i:i_end], segments[j:j_end])
    return ssm


def find_dense_segments(novelty, hop_sec, segment_duration=60, min_peaks=3, overlap_ratio=0.5):
    """이벤트 밀집 1분 구간 추천"""
    distance_frames = max(1, int(3 / hop_sec))  # 3초 간격
    peaks, properties = find_peaks(novelty, height=0.1, distance=distance_frames)
    
    frames_per_segment = int(segment_duration / hop_sec)
    step = int(frames_per_segment * (1 - overlap_ratio))
    candidates = []
    
    for start in range(0, len(novelty) - frames_per_segment, step):
        end = start + frames_per_segment
        segment_peaks = peaks[(peaks >= start) & (peaks < end)]
        
        if len(segment_peaks) >= min_peaks:
            start_sec = start * hop_sec
            end_sec = end * hop_sec
            avg_height = np.mean(novelty[segment_peaks])
            
            candidates.append({
                'start': int(start_sec),
                'end': int(end_sec),
                'peak_count': len(segment_peaks),
                'avg_novelty': round(float(avg_height), 3),
                'start_fmt': f'{int(start_sec//60)}:{int(start_sec%60):02d}',
                'end_fmt': f'{int(end_sec//60)}:{int(end_sec%60):02d}',
            })
    
    # 피크 수 > 평균 novelty 순으로 정렬
    candidates.sort(key=lambda x: (x['peak_count'], x['avg_novelty']), reverse=True)
    
    # 겹치는 구간 제거
    selected = []
    for cand in candidates:
        overlap = False
        for sel in selected:
            if cand['start'] < sel['end'] and cand['end'] > sel['start']:
                overlap = True
                break
        if not overlap:
            selected.append(cand)
    
    return selected


def main():
    parser = argparse.ArgumentParser(description="게임 영상 이벤트 밀집 구간 추천")
    parser.add_argument("youtube_id", help="YouTube 영상 ID (예: EtoTzyT8GbI)")
    parser.add_argument("--duration", type=int, default=60, help="구간 길이 (초, 기본 60)")
    parser.add_argument("--window", type=float, default=5.0, help="SSM 윈도우 크기 (초, 기본 5)")
    parser.add_argument("--hop", type=float, default=1.0, help="SSM 홉 크기 (초, 기본 1)")
    parser.add_argument("--min_peaks", type=int, default=3, help="최소 피크 수 (기본 3)")
    parser.add_argument("--top", type=int, default=10, help="추천 구간 수 (기본 10)")
    parser.add_argument("--output_json", type=str, default=None, help="JSON 저장 경로 (기본: {youtube_id}_segments.json)")
    parser.add_argument("--video_id_prefix", type=str, default="segment", help="JSON의 video_id 접두사")
    args = parser.parse_args()
    
    # 1. 오디오 다운로드
    audio_path = download_audio(args.youtube_id)
    
    # 2. MFCC 추출
    print("[2/5] MFCC 추출 중...")
    y, sr = librosa.load(audio_path, sr=22050)
    total_sec = len(y) / sr
    print(f"   -> 총 길이: {int(total_sec//60)}분 {int(total_sec%60)}초")

    mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
    mfcc_T = mfcc.T
    del y  # 오디오 원본 메모리 해제

    # 3. SSM 구성
    print("[3/5] SSM 구성 중...")
    hop_sec = args.hop
    window_sec = args.window
    frames_per_sec = sr / 512
    win_frames = int(window_sec * frames_per_sec)
    hop_frames = int(hop_sec * frames_per_sec)

    segments = []
    for start in range(0, len(mfcc_T) - win_frames, hop_frames):
        segment = mfcc_T[start:start + win_frames].mean(axis=0)
        segments.append(segment)
    segments = np.array(segments)
    del mfcc_T  # MFCC 메모리 해제

    # 메모리 사용량 추정 및 경고
    n = len(segments)
    ssm_mb = n * n * 4 / (1024 ** 2)  # float32 기준
    print(f"   -> 세그먼트 수: {n}, 예상 SSM 메모리: {ssm_mb:.0f} MB")
    if ssm_mb > 2048:
        print(f"   [경고] SSM이 너무 큼 ({ssm_mb:.0f} MB). --hop을 키우거나 --window를 조정하세요.")
        sys.exit(1)

    use_chunked = ssm_mb > 500
    if use_chunked:
        print(f"   -> 청크 방식으로 SSM 계산 (메모리 절약)")
        ssm = compute_ssm_chunked(segments)
    else:
        ssm = cosine_similarity(segments).astype(np.float32)
    del segments  # 세그먼트 메모리 해제
    print(f"   -> SSM shape: {ssm.shape}")
    
    # 4. Novelty score 계산
    print("[4/5] Novelty score 계산 중...")
    novelty = compute_novelty(ssm, kernel_size=16)
    del ssm  # SSM 메모리 해제
    
    # 5. 밀집 구간 추천
    print(f"[5/5] 이벤트 밀집 {args.duration}초 구간 추천 중...\n")
    candidates = find_dense_segments(
        novelty, hop_sec, 
        segment_duration=args.duration,
        min_peaks=args.min_peaks
    )
    
    top_candidates = sorted(candidates[:args.top], key=lambda x: x['start'])
    
    # 결과 출력
    print("=" * 60)
    print(f"  추천 구간 (상위 {len(top_candidates)}개, 피크 {args.min_peaks}개 이상)")
    print("=" * 60)
    for i, seg in enumerate(top_candidates):
        print(f"  {i+1:2d}. {seg['start_fmt']} ~ {seg['end_fmt']}  "
              f"| 피크 {seg['peak_count']}개 | avg novelty {seg['avg_novelty']}")
    print("=" * 60)
    
    # JSON 저장
    output_path = args.output_json or f"{args.youtube_id}_segments.json"
    json_data = []
    for i, seg in enumerate(top_candidates):
        json_data.append({
            "video_id": f"{args.video_id_prefix}_{i+1:02d}",
            "youtube_id": args.youtube_id,
            "start": seg['start'],
            "end": seg['end'],
            "peak_count": seg['peak_count'],
            "avg_novelty": seg['avg_novelty'],
        })

    with open(output_path, 'w', encoding='utf-8') as f:
        json.dump(json_data, f, indent=2, ensure_ascii=False)
    print(f"\n-> JSON 저장 완료: {output_path}")
    
    # 정리
    if os.path.exists(audio_path):
        os.remove(audio_path)
    
    print("\n완료!")


if __name__ == "__main__":
    main()
```
