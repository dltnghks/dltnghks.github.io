## 풀이 메모
```c++
#include <iostream>

using namespace std;

long long tree[100000 * 4 + 1];
long long arr[100001];

long long Sum(int cur, int left, int right, int targetL, int targetR){
    if(targetL <= left && right <= targetR){
        return tree[cur];
    }

    if(targetL > right || left > targetR){
        return 0;
    }

    int mid = (left + right)/2;
    return Sum(cur * 2, left, mid, targetL, targetR) + Sum(cur * 2 + 1, mid+1, right, targetL, targetR);
}

void Change(int cur, int left, int right, int target, long long diff){
    if(left == target && right == target){
        tree[cur] += diff;
        return;
    }

    if(left > target || right < target){
        return;
    }

    int mid = (left + right) / 2;
    tree[cur] += diff;
    if(target <= mid){
        Change(cur * 2, left, mid, target, diff);
    }
    else if(target > mid){
        Change(cur * 2 + 1, mid+1, right, target, diff);
    }
}

long long MakeTree(int cur, int left, int right){
    if(left >= right){
        tree[cur] = arr[left];
        return tree[cur];
    }

    int mid = (left+right)/2;
    tree[cur] = MakeTree(cur * 2, left, mid) + MakeTree(cur * 2 + 1, mid+1, right);
    return tree[cur];
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    // N개의 정수
    int n, q;
    cin >> n >> q;

    for(int i = 1; i <= n; i++){
        cin >> arr[i];
    }

    MakeTree(1, 1, n);

    // x~y번째 숫자 합?
    // 답은 00 입니다. a 번째 수를 b로 바꾼다.
    for(int i = 0; i < q; i++){
        int x, y, a;
        long long b;
        cin >> x >> y >> a >> b;
        cout << Sum(1, 1, n, x, y) << '\n';

        long long diff = b - arr[a];
        arr[a] = b;
        Change(1, 1, n, a, diff);
    }

    return 0;
}

```

## 회고
- 펜윅 트리로도 구현할 수 있나? 이거 풀이 정확하게 이해하고 한 번 구현해봐야겠는데
```c++
#include <algorithm>
#include <iostream>

using namespace std;

long long tree[100001];
long long arr[100001];

void Update(int idx, int n, long long diff){
    while(idx <= n){
        tree[idx] += diff;
        idx += idx & -idx;
    }
}

long long PrefixSum(int idx){
    long long sum = 0;
    while(idx > 0){
        sum += tree[idx];
        idx -= idx & -idx;
    }
    return sum;
}

long long RangeSum(int left, int right){
    return PrefixSum(right) - PrefixSum(left - 1);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    int n, q;
    cin >> n >> q;

    for(int i = 1; i <= n; i++){
        cin >> arr[i];
        Update(i, n, arr[i]);
    }

    for(int i = 0; i < q; i++){
        int x, y, a;
        long long b;
        cin >> x >> y >> a >> b;

        if(x > y){
            swap(x, y);
        }
        cout << RangeSum(x, y) << '\n';

        long long diff = b - arr[a];
        arr[a] = b;
        Update(a, n, diff);
    }

    return 0;
}

```
