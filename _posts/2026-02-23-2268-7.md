## 풀이 메모
- 병목 포인트는 Modify 구현
	- 처음에는 아래까지 쭉 내려갔다가 다시 올라오는 식으로 구현
	- 지금은 diff를 미리 구한 후 한 번 내려가면 끝나도록 수정정

```c++
#include <iostream>
#include <vector>

using namespace std;

vector<int> arr;
vector<long long> tree;

long long Sum(int cur, int left, int right, int targetL, int targetR){
    // 포함 조건
    if(targetL <= left && right <= targetR){
        return tree[cur];
    }

    // 미포함 조건으로 구분
    if (right < targetL || targetR < left) {
        return 0;
    }


    int mid = (left+right) / 2;
    return Sum(cur * 2, left, mid, targetL, targetR) + Sum(cur * 2 + 1, mid+1, right, targetL, targetR);

}

void Modify(int cur, int left, int right, int index, long long diff){
    if(index == left && index == right){
        tree[cur] += diff;
        return;
    }

    // 범위를 벗어났을 때
    if(right < index || left > index){
        return;
    }

    if(left <= index && index <= right){
        int mid = (left+right) / 2;
        tree[cur] += diff;
        Modify(cur * 2, left, mid, index, diff);
        Modify(cur * 2 + 1, mid+1, right, index, diff);
    }

    return;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    
    int n, m;
    cin >> n >> m;

    tree = vector<long long>(4*n+1, 0);
    arr = vector<int>(n+1, 0);

    for(int i = 0; i < m; i++){
        int oper, a;
        cin >> oper >> a;
        if(oper == 0){
            // Sum
            int b;
            cin >> b;

            if (a > b) swap(a, b);

            cout << Sum(1, 1, n, a, b) << '\n';
        }else if(oper == 1){
            // Modify
            long long k;
            cin >> k;

            long long diff = k - arr[a];
            arr[a] = k;
            Modify(1, 1, n, a, diff);
        }
    }
    
    return 0;
}

```
