## MFCC 추출 과정
MFCC는 오디오 신호에서 단계적으로 정보를 변환해가는 과정
![image-9](/assets/img/posts/2026-03-19-03-19-progress/image-9-1edf68c1.png)
**1단계 - 프레임 분할:** 오디오를 약 0.02초(20ms)단위로 잘게 쪼갠다. 이 짧은 구간 안에서는 소리가 거의 일정하다고 가정.

**2단계 - FFT:** 각 프레임에 FFT를 적용해서 "이 0.02초 안에 어떤 주파수 성분이 얼마나 들어있는가"를 알아낸다. waveform은 시간축이었는데, FFT를 거치면서 주파수축으로 바뀐다.
예를 들어 총소리는 넓은 주파수 대역에 에너지가 퍼져있고, BGM의 저음 베이스는 낮은 주파수에 에너지가 몰려있다.

**3단계 - Mel 필터:** 사람 귀는 저주파(낮은 소리)에 민감하고 고주파(높은 소리)는 잘 구분 못한다. Mel 필터는 이 특성을 반영해서 저주파는 촘촘하게, 고주파는 뭉뚱그려서 본다. 불필요한 세부 정보를 버리고 "사람이 인지하는 차이"에 집중하는 것이다.

**4단계 - 로그:** 소리 에너지는 범위가 어마어마하게 넓어서(속삭임 vs 폭발음), 로그를 씌워서 범위를 압축한다. 데시벨(dB)을 쓰는 것과 같은 원리다. 

**5단계 - DCT:** 마지막으로 DCT를 적용해서 상위 13개 계수만 뽑는다. 이게 해당 프레임의 "소리 지문"역할을 하는 MFCC다.

## Python으로 오디오 불러오기 + waveform 시각화
```python
import librosa
import librosa.display
import matplotlib.pyplot as plt
import numpy as np

# 오디오 로드
y, sr = librosa.load('audio.wav', sr=22050)

# 기본 정보 출력
print(f'샘플레이트: {sr}')
print(f'총 샘플 수: {len(y)}')

librosa.display.waveshow(y, sr=sr)
plt.title('Waveform')
plt.xlabel('Time (s)')
plt.ylabel('Amplitude')
plt.tight_layout()
plt.savefig('waveform.png', dpi=150)
plt.show()
```

## Macro Meso SSM
```python
import librosa
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics.pairwise import cosine_similarity

# 전체 오디오 로드
y, sr = librosa.load('audio.wav', sr=22050)
mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
mfcc_T = mfcc.T  # (프레임 수, 13)

print(f'전체 프레임 수: {mfcc_T.shape[0]}')

def make_ssm(features, window_sec, hop_sec, sr=22050, hop_length=512):
    """윈도우 단위로 평균내서 SSM 생성"""
    frames_per_sec = sr / hop_length
    win_frames = int(window_sec * frames_per_sec)
    hop_frames = int(hop_sec * frames_per_sec)
    
    segments = []
    for start in range(0, len(features) - win_frames, hop_frames):
        segment = features[start:start + win_frames].mean(axis=0)
        segments.append(segment)
    
    segments = np.array(segments)
    print(f'윈도우 {window_sec}s, 홉 {hop_sec}s → 세그먼트 수: {len(segments)}')
    return cosine_similarity(segments)

# Macro SSM (20초 윈도우, 5초 홉)
ssm_macro = make_ssm(mfcc_T, window_sec=20, hop_sec=5)

# Meso SSM (3초 윈도우, 1초 홉)  
ssm_meso = make_ssm(mfcc_T, window_sec=3, hop_sec=1)

# 시각화
fig, axes = plt.subplots(1, 2, figsize=(16, 7))

axes[0].imshow(ssm_macro, cmap='magma', origin='lower')
axes[0].set_title('Macro SSM (20s window)')
axes[0].set_xlabel('Segment')
axes[0].set_ylabel('Segment')

axes[1].imshow(ssm_meso, cmap='magma', origin='lower')
axes[1].set_title('Meso SSM (3s window)')
axes[1].set_xlabel('Segment')
axes[1].set_ylabel('Segment')

plt.tight_layout()
plt.savefig('ssm_comparison.png', dpi=150)
plt.show()
```

## Checkboard
```python
import numpy as np
import matplotlib.pyplot as plt
import librosa
from sklearn.metrics.pairwise import cosine_similarity
from scipy.signal import find_peaks


# 오디오 로드 + MFCC 추출
y, sr = librosa.load('../data/audio/battle_ground_play_audio.wav', sr=22050)
mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)

# MFCC 전치: (프레임 수, 13) 형태로 변환
# 각 행이 한 시점의 소리 특성 벡터가 됨
mfcc_T = mfcc.T

print(f'프레임 수: {mfcc_T.shape[0]}')
print(f'각 프레임의 특성 벡터 차원: {mfcc_T.shape[1]}')

def make_ssm(features, window_sec, hop_sec, sr=22050, hop_length=512):
    """윈도우 단위로 평균내서 SSM 생성"""
    frames_per_sec = sr / hop_length
    win_frames = int(window_sec * frames_per_sec)
    hop_frames = int(hop_sec * frames_per_sec)

    segments = []
    for start in range(0, len(features) - win_frames, hop_frames):
        segment = features[start:start + win_frames].mean(axis=0)
        segments.append(segment)

    segments = np.array(segments)
    print(f'윈도우 {window_sec}s, 홉 {hop_sec}s → 세그먼트 수: {len(segments)}')
    return cosine_similarity(segments)


def checkerboard_kernel(size):
    """Checkerboard 커널 생성"""
    kernel = np.ones((size, size))
    for i in range(size):
        for j in range(size):
            if (i < size//2 and j < size//2) or (i >= size//2 and j >= size//2):
                kernel[i, j] = 1
            else:
                kernel[i, j] = -1
    return kernel

def compute_novelty(ssm, kernel_size=16):
    """SSM 대각선을 따라 novelty score 계산"""
    kernel = checkerboard_kernel(kernel_size)
    half = kernel_size // 2
    n = ssm.shape[0]
    novelty = np.zeros(n)
    
    for i in range(half, n - half):
        # 대각선 위 커널 크기만큼 영역 추출
        patch = ssm[i-half:i+half, i-half:i+half]
        novelty[i] = np.sum(patch * kernel)
    
    # 0~1로 정규화
    novelty = np.maximum(novelty, 0)
    if novelty.max() > 0:
        novelty /= novelty.max()
    
    return novelty

# Macro SSM (20초 윈도우, 5초 홉)
ssm_macro = make_ssm(mfcc_T, window_sec=20, hop_sec=5)

# Meso SSM (3초 윈도우, 1초 홉)  
ssm_meso = make_ssm(mfcc_T, window_sec=3, hop_sec=1)
# 시각화
fig, axes = plt.subplots(3, 2, figsize=(18, 12))

macro_hop = 5   # Macro: 5초 hop
meso_hop = 1    # Meso: 1초 hop

# 왼쪽: Macro SSM
for idx, ks in enumerate([8, 16, 32]):
    novelty = compute_novelty(ssm_macro, kernel_size=ks)
    peaks, _ = find_peaks(novelty, height=0.3, distance=5)
    time_axis = np.arange(len(novelty)) * macro_hop / 60
    peak_times = peaks * macro_hop / 60
    
    axes[idx][0].plot(time_axis, novelty, color='coral')
    axes[idx][0].plot(peak_times, novelty[peaks], 'v', color='red', markersize=10)
    axes[idx][0].set_title(f'Macro | kernel={ks} → Boundary {len(peaks)}개')
    axes[idx][0].set_ylabel('Novelty')
    axes[idx][0].set_xlabel('Time (min)')

# 오른쪽: Meso SSM
for idx, ks in enumerate([16, 32, 64]):
    novelty = compute_novelty(ssm_meso, kernel_size=ks)
    peaks, _ = find_peaks(novelty, height=0.3, distance=10)
    time_axis = np.arange(len(novelty)) * meso_hop / 60
    peak_times = peaks * meso_hop / 60
    
    axes[idx][1].plot(time_axis, novelty, color='teal')
    axes[idx][1].plot(peak_times, novelty[peaks], 'v', color='red', markersize=10)
    axes[idx][1].set_title(f'Meso | kernel={ks} → Boundary {len(peaks)}개')
    axes[idx][1].set_ylabel('Novelty')
    axes[idx][1].set_xlabel('Time (min)')


plt.tight_layout()
plt.savefig('kernel_size_comparison.png', dpi=150)
plt.show()
```
